     def parser_documents(self, documents: List[Any]) -> str:
         """把检索到的相似案例组织为 LLM prompt 上下文（简单拼接）。"""
         pieces = []
         for i, doc in enumerate(documents):
             text = getattr(doc, "page_content", "") or doc.get("page_content", "")
             # meta_result = ""
             try:
                 meta_result = doc.metadata.get("result", "")
             except Exception:
                 meta_result = doc.get("metadata", {}).get("result", "")
             pieces.append(f"案例{i + 1}: {text}判决结果：{meta_result}")
         return "\n\n".join(pieces)

     def dashscope_call_get_content(self, messages: List[Dict[str, str]], model: str) -> str:
         """
         统一用 dashscope Generation.call 发请求并安全返回字符串 content。
         返回字符串（如果失败返回空字符串）。
         """
         api_key = DASHSCOPE_API_KEY
         if not api_key:
             logger.error("环境变量 DASHSCOPE_API_KEY 未设置，dashscope 调用会失败。")

                 api_key=api_key,
                 model=model,
                 messages=messages,
                 result_format="message",
                 stream=False,
                 enable_thinking=False

             # 常见返回解析

                 out = resp.get("output", {})

             else:

         except Exception as e:

             return ""

     def merge_result(self, all_result: List[Dict[str, Any]]) -> Dict[str, Any]:
         """
         对多次模型输出的刑期/罚金做平均并合并死刑/无期标志。
         返回一个标准化的字典，数值字段为 int。
         """
         if not all_result:
             return {"罪名": [], "罚金": 0, "是否死刑": False, "是否无期": False, "有期徒刑": 0, "犯罪嫌疑人": []}

         death = any(bool(r.get("是否死刑")) for r in all_result)
         imprisonment = any(bool(r.get("是否无期")) for r in all_result)

         months_list = []
         fines = []
         for r in all_result:
             if not r.get("是否死刑") and not r.get("是否无期"):
                 try:
                     months_list.append(float(r.get("有期徒刑", 0)))
                 except Exception:
                     pass
             try:
                 fines.append(float(r.get("罚金", 0)))
             except Exception:
                 pass

         avg_months = int(round(sum(months_list) / len(months_list))) if months_list else 0
         avg_fine = int(round(sum(fines) / len(fines))) if fines else 0

         merged = all_result[0].copy()
         merged["有期徒刑"] = avg_months
         merged["罚金"] = avg_fine
         merged["是否死刑"] = death
         merged["是否无期"] = imprisonment

         # 规范字段
         if "犯罪嫌疑人" not in merged:
             merged["犯罪嫌疑人"] = []
         if "罪名" not in merged:
             merged["罪名"] = []

         return merged

     def accusation_json2str(self, data: Dict[str, Any]) -> str:
         """把判决 JSON 转为一句描述性的字符串（便于拼接给司法解释模型）。"""

         result += "罚金:" + str(int(data.get("罚金", 0))) + "元，"
         suspects = data.get("犯罪嫌疑人", [])
         if not isinstance(suspects, list):
             suspects = [str(suspects)]
         result += "罪犯:" + "，".join(suspects) + "，"
         if data.get("是否死刑"):
             result += "刑期：死刑"
         elif data.get("是否无期"):
             result += "刑期：无期徒刑"
         else:

             months = int(round(float(data.get("有期徒刑", 0))))
             result += f"刑期：有期徒刑 {months} 个月"
         return result

     def parser_documents(self, documents: List[Any]) -> str:

     def merge_result(self, all_result: List[Dict[str, Any]]) -> Dict[str, Any]:
         """
         对多次模型输出的刑期/罚金做平均并合并死刑/无期标志。
         返回一个标准化的字典，数值字段为 int。
         """
         if not all_result:
             return {"罪名": [], "罚金": 0, "是否死刑": False, "是否无期": False, "有期徒刑": 0, "犯罪嫌疑人": []}

         death = any(bool(r.get("是否死刑")) for r in all_result)
         imprisonment = any(bool(r.get("是否无期")) for r in all_result)

         months_list = []
         fines = []
         for r in all_result:
             if not r.get("是否死刑") and not r.get("是否无期"):
                 try:
                     months_list.append(float(r.get("有期徒刑", 0)))
                 except Exception:
                     pass
             try:
                 fines.append(float(r.get("罚金", 0)))
             except Exception:
                 pass

         avg_months = int(round(sum(months_list) / len(months_list))) if months_list else 0
         avg_fine = int(round(sum(fines) / len(fines))) if fines else 0

         merged = all_result[0].copy()
         merged["有期徒刑"] = avg_months
         merged["罚金"] = avg_fine
         merged["是否死刑"] = death
         merged["是否无期"] = imprisonment

         # 规范字段
         if "犯罪嫌疑人" not in merged:
             merged["犯罪嫌疑人"] = []
         if "罪名" not in merged:
             merged["罪名"] = []

         return merged

         suspects = data.get("犯罪嫌疑人", [])
         if not isinstance(suspects, list):
             suspects = [str(suspects)]
         result += "罪犯:" + "，".join(suspects) + "，"
         if data.get("是否死刑"):
             result += "刑期：死刑"
         elif data.get("是否无期"):
             result += "刑期：无期徒刑"
         else:

             months = int(round(float(data.get("有期徒刑", 0))))
             result += f"刑期：有期徒刑 {months} 个月"

**7️⃣ 创建 fix_parser**

```python
fix_parser = OutputFixingParser.from_llm(
    parser=parser,
    llm=llm_wrapper

```

🧠 这一步是全项目最值钱的地方

它解决什么问题？

> ❌ LLM 经常返回不合法 JSON
> ✅ OutputFixingParser 会自动：

1. 发现 JSON 不合法
2. 把“坏输出”喂给 LLM
3. 要求它修成合法 JSON
4. 再 parse 一次

📌 面试一句话秒杀：

> 通过 OutputFixingParser 显著提升结构化输出的稳定性，避免模型输出漂移。

## 4 测试集评估模型

```py
# 13_model_evaluation.py
def get_month(data):
    if data["是否死刑"]:
        return 12 * 100
    if data["是否无期"]:
        return 12 * 50
    return data["有期徒刑"]


with open(r"..\test_result", encoding="utf-8") as f:
    lines = [eval(s.strip()) for s in f.readlines()]

std = 0.25
right_charge = 0
right_person = 0
right_fine = 0
right_time = 0
count = 0
for data in lines:
    try:
        label_output = data[0]
        predict_output = data[1]
        # print(label_output, predict_output)
        if label_output["罪名"][0] in predict_output["罪名"]:
            right_charge += 1
        if str(label_output["犯罪嫌疑人"]) == str(predict_output["犯罪嫌疑人"]):
            right_person += 1
        if predict_output["罚金"] >= (1 - std) * label_output["罚金"] and predict_output["罚金"] <= (1 + std) * label_output["罚金"]:
            right_fine += 1
        fact_time = get_month(label_output)
        predict_time = get_month(predict_output)

        if predict_time >= (1 - std) * fact_time and predict_time <= (1 + std) * fact_time:
            right_time += 1

        count += 1
    except:
        continue

print("罪名准确率", right_charge / count)
print("罪犯准确率", right_person / count)
print("罚金准确率", right_fine / count)
print("坐牢时间准确率", right_time / count)
```
