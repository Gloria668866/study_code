```py
# 导入json模块
import json
# 从json_output模块导入parser
from  json_output import parser
# 从transformers模块导入AutoTokenizer和AutoConfig
from transformers import AutoTokenizer,AutoConfig
# 从tqdm模块导入tqdm
from tqdm import tqdm
import os, sys

# 定义处理函数，参数为feature
def process(feature):
    # 构建提示信息，包含输入内容和判决结果
    prompt = feature["input"] + "\n\n\n" + "判决结果:"

    # 构建消息列表，包含系统角色和用户角色的内容
    messages = [{"role": "system", "content": "你是一个法律顾问，根据内容预测所犯罪名，罚金，刑期"},
                {"role": "user", "content": prompt}]
    # 使用tokenizer将消息列表转换为文本  生成聊天模板
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    print(text)
    print(type(text))
    # 将输出结果转换为JSON格式
    output = json.dumps(feature["output"], ensure_ascii=False)
    print(output)
    # 将输入文本编码为token ID列表
    input_ids = tokenizer.encode(text, truncation=True)
    print(input_ids)
    print(len(input_ids))  # 输入文本的长度230
    # 将输出结果编码为token ID列表，并添加结束符
    output_ids = tokenizer.encode(output, truncation=True) + [config.eos_token_id]
    print(output_ids)
    print(len(output_ids))  # 输出结果的长度44
    # 将输入和输出的token ID列表合并
    ids = input_ids + output_ids[0:-1]
    # 如果合并后的token ID列表长度超过1000
    if len(ids) > 1000:
        # 返回None
        return None,None
    # 返回输入和输出的token ID列表
    return input_ids, output_ids


if __name__ == '__main__':

    # 定义模型路径
    model_name="E:\\code\\qwen2.5-1.5B-Instruct"
    # 加载tokenizer
    tokenizer = AutoTokenizer.from_pretrained(model_name, trust_remote_code=True)
    # 加载模型配置
    config = AutoConfig.from_pretrained(model_name, trust_remote_code=True,
                                        device_map='balanced')
    # 定义训练数据文件路径
    train_data_path = r"E:\baidu\刑法大模型\law\law\data1\train_data.jsonl"
    # 读取训练数据文件
    with open(train_data_path, encoding="utf-8")  as f:
        # 将文件内容解析为JSON对象列表
        dataset = [json.loads(s.strip()) for s in f.readlines()]
    # print(dataset[0])

    # 定义结果列表
    result = []
    # 使用tqdm显示进度条
    for feature in tqdm(dataset):
        # 处理每个数据项
        input_ids, output_ids = process(feature)
        # 如果处理结果为None，跳过当前数据项
        if input_ids is None:
            continue
        # 将处理结果添加到结果列表中
        result.append(json.dumps({"input_ids": [input_ids, output_ids]}))
```

```py
def data_collator(features) -> dict:
    max_seq_length = -1
    input_list = []

    # 规定输入和输出的加起来的尺寸在 0 -> 1000 之间

    for feature in features:
        input_ids, output_ids = feature["input_ids"]
        # 将输入输出 加起来 得到一个完整的序列  最后一个是结束符 不要
        ids = input_ids + output_ids[0:-1]
        # 限制最大长度
        if len(ids) > 1000:
            continue
        max_seq_length = max(max_seq_length, len(ids))
        tmp_ids.append([input_ids, output_ids])

    # print("最大长度", max_seq_length)
    # print(tmp_ids)
    # 将数据处理成输入和输出的形式

        labels = labels + (max_seq_length - len(labels)) * [-100]

    return {
```

## 6 构建罪名向量数据库

罪名向量数据库的意义是：

- 罪名预测是高风险任务，不能黑箱操作，做到可解释，可回溯，可质证
- 大模型容易混淆相似罪名，容易自信胡说，罪名向量数据库可以把模型的输出限制在真实存在的罪名集合内。
- 从工程和合规角度来看，罪名预测阶段构建罪名向量数据库并非算法上的必需条件，但在真实司法系统中是非常关键的一层。它承担了罪名候选召回与语义对齐的职责，既约束了大模型输出范围，降低幻觉风险，又为后续构成要件比对和量刑分析提供了稳定入口，是罪名辅助认定能够落地的基础设施。

### 6.1 自定义编码类

**Embedding.py**

```python
from langchain.embeddings.base import Embeddings
from typing import List
import torch
from transformers import BertTokenizer, BertModel


def l2_normalize(vector):
    """对向量进行 L2 归一化"""
    norm = torch.norm(vector)
    if norm.item() == 0:
        return vector
    return vector / norm


class CustomEmbeddings(Embeddings):
    """使用 BERT 作为向量生成器的自定义 Embedding 类"""

    def __init__(self, model_path: str = r"E:\code\bge\bge_recall", device: str = None):
        self.device = device or ("cuda:0" if torch.cuda.is_available() else "cpu")

        # 加载 tokenizer 和 model
        self.tokenizer = BertTokenizer.from_pretrained(model_path)
        self.model = BertModel.from_pretrained(model_path).to(self.device)

        # 设置 eval 模式，关闭 dropout
        self.model.eval()

    def embed_documents(self, texts: List[str]) -> List[List[float]]:
        """将文档列表转为向量"""
        return [self.embed_query(text) for text in texts]

    def embed_query(self, text: str) -> List[float]:
        """将单个文本编码为向量"""

        # Tokenize 文本
        tokens = self.tokenizer(
            text,
            padding=True,
            truncation=True,
            return_tensors="pt"
        ).to(self.device)
        # print(tokens)
        # 关闭梯度，提高推理速度、减少显存
        with torch.no_grad():
            outputs = self.model(**tokens)
        # print(outputs.last_hidden_state.shape)  # torch.Size([1, 13, 512])
        # print(outputs.pooler_output.shape)  # ([1, 512])
        # outputs[1] = pooled_output（CLS向量）
        # 为什么 BERT 要使用 pooler_output（CLS 向量）来做句向量?
        # print(outputs)
        pooled = outputs.pooler_output.squeeze(0)
        # print(pooled.shape)  # torch.Size([512])
        # L2 normalize
        normalized = l2_normalize(pooled)
        # print(normalized)
        return normalized.cpu().tolist()


# 使用示例
model_path = r"E:\code\bge\bge_recall"
emb = CustomEmbeddings(model_path=model_path)

# 示例
vec = emb.embed_query("你好，我是大模型工程师")
print(len(vec), vec[:10])

```

- **问题1：为什么 BERT 要使用 pooler_output（CLS 向量）来做句（文本）向量？**
  - BERT 使用 `pooler_output`（即 `[CLS]` 向量）作为句向量，是因为在预训练阶段 `[CLS]` 被设计为聚合整句语义的全局表示，并在下游句级任务中通过监督信号进行优化

## 8 LoRA微调分类模型

```py
# 导入所需的库和模块
import json
import os
import torch
import torch.nn as nn
from dataclasses import dataclass, field
from transformers import (
    TrainingArguments, 
    Trainer, 
    AutoTokenizer, 
    AutoConfig, 
    AutoModelForSequenceClassification
)
from peft import get_peft_model, LoraConfig, TaskType
from json_output import parser

class LegalTextClassifierTrainer:
    """法律文本分类训练器"""
    
    def __init__(self, model_path, accusation_map_path):
        """
        初始化训练器
        
        Args:
            model_path: 预训练模型路径
            accusation_map_path: 罪名映射文件路径
        """
        self.model_path = model_path
        self.accusation_map_path = accusation_map_path
        self._setup_environment()
        self._load_resources()
        
    def _setup_environment(self):
        """设置环境配置"""
        # 禁用OneDNN优化（如需要）
        # os.environ["TF_ENABLE_ONEDNN_OPTS"] = "0"
        pass
        
    def _load_resources(self):
        """加载模型和罪名映射"""
        # 加载罪名ID映射
        with open(self.accusation_map_path, encoding="utf-8") as f:
            self.accusation_id = json.load(f)
        
        # 初始化分词器和模型
        self.tokenizer = AutoTokenizer.from_pretrained(
            self.model_path, 
            trust_remote_code=True
        )
        
        self.model = AutoModelForSequenceClassification.from_pretrained(
            self.model_path,
            load_in_8bit=False,
            trust_remote_code=True,
            device_map=None,
            num_labels=len(self.accusation_id)  # 分类数量等于罪名数量
        )
        
        # 配置pad token
        self.model.config.pad_token_id = self.tokenizer.pad_token_id
        
    def data_collator(self, features) -> dict:
        """
        数据整理函数，处理批次数据
        
        Args:
            features: 批次特征列表
            
        Returns:
            dict: 整理后的模型输入
        """
        max_seq_length = -1
        input_list = []
        labels_list = []
        
        # 第一遍：找出批次中最长序列长度
        for feature in features:
            input_ids = feature["input_ids"]["input"]
            if len(input_ids) > 1000:  # 过滤过长序列
                continue
            max_seq_length = max(len(input_ids), max_seq_length)
        
        # 第二遍：填充序列并构建张量
        for feature in features:
            input_ids = feature["input_ids"]["input"]
            if len(input_ids) > 1000:
                continue
                
            label = feature["input_ids"]["output"]
            
            # 右侧填充序列到最大长度
            padding_length = max_seq_length - len(input_ids)
            padded_ids = input_ids + [self.tokenizer.pad_token_id] * padding_length
            
            input_list.append(torch.LongTensor(padded_ids))
            labels_list.append(torch.LongTensor([label]))
        
        # 堆叠张量
        input_ids_tensor = torch.stack(input_list)
        labels_tensor = torch.stack(labels_list)
        
        return {
            "input_ids": input_ids_tensor,
            "labels": labels_tensor,
        }
    
    def setup_lora_config(self):
        """配置LoRA参数"""
        lora_config = LoraConfig(
            r=16,                      # LoRA矩阵的秩
            lora_alpha=16,             # LoRA缩放因子
            target_modules=["q_proj", "v_proj"],  # 应用LoRA的注意力模块
            lora_dropout=0.1,          # Dropout概率
            bias="none",               # 不训练偏置项
            task_type=TaskType.SEQ_CLS # 序列分类任务
        )
        return lora_config
    
    def setup_training_args(self):
        """配置训练参数"""
        training_args = TrainingArguments(
            output_dir="./qwen_lora_output",        # 输出目录
            learning_rate=2e-4,                     # 学习率
            per_device_train_batch_size=4,          # 每设备批次大小
            gradient_accumulation_steps=4,          # 梯度累积步数
            num_train_epochs=3,                     # 训练轮次
            weight_decay=0.01,                      # 权重衰减
            logging_dir="./logs",                   # 日志目录
            logging_steps=10,                       # 日志记录频率
            save_strategy="steps",                  # 保存策略
            save_steps=30,                          # 保存频率
            fp16=True,                              # 使用混合精度训练
            push_to_hub=False,                      # 不推送到Hugging Face Hub
            report_to="tensorboard",                # 报告到TensorBoard
            save_total_limit=2,                     # 最多保存2个检查点
        )
        return training_args
    
    def load_dataset(self, data_path):
        """加载训练数据集"""
        with open(data_path, encoding="utf-8") as f:
            dataset = [{"input_ids": json.loads(line.strip())} for line in f]
        return dataset
    
    def train(self, data_path):
        """执行训练流程"""
        print("开始法律文本分类模型训练...")
        
        # 配置LoRA
        lora_config = self.setup_lora_config()
        self.model = get_peft_model(self.model, lora_config)
        
        print("LoRA模型配置完成:")
        print(self.model)
        
        # 解冻分类头参数（LoRA默认会冻结所有层）
        for name, param in self.model.score.named_parameters():
            param.requires_grad = True
            print(f"解冻参数层: {name}")
        
        # 加载训练数据
        dataset = self.load_dataset(data_path)
        print(f"加载训练样本: {len(dataset)} 条")
        
        # 配置训练参数
        training_args = self.setup_training_args()
        
        # 初始化训练器
        trainer = Trainer(
            model=self.model,
            args=training_args,
            train_dataset=dataset,
            data_collator=self.data_collator,
        )
        
        # 启动训练
        print("开始模型训练...")
        trainer.train()
        
        # 保存模型
        self.model.save_pretrained("./qwen_lora_model")
        torch.save(self.model.score.state_dict(), "score_weights.pt")
        print("模型训练完成并保存!")
        
        return trainer

def main():
    """主函数"""
    # 配置路径

    ACCUSATION_MAP_PATH = "accusation_id"
    TRAIN_DATA_PATH = "data\\train_data_classify_token"
    
    # 初始化训练器
    trainer = LegalTextClassifierTrainer(MODEL_PATH, ACCUSATION_MAP_PATH)

    trainer.train(TRAIN_DATA_PATH)

if __name__ == "__main__":
    main()
```

**1. 文件头与依赖（import）**

脚本开始导入了需要的库：

- `json/os/torch`：基本的文件处理与 PyTorch 张量操作。
- `transformers` 的 `TrainingArguments/Trainer/AutoTokenizer/AutoModelForSequenceClassification`：用于加载预训练模型、分词器，并用 HuggingFace 的训练器进行训练。
- `peft` 的 `get_peft_model/LoraConfig/TaskType`：用于把 LoRA（低秩适配器）加到大模型上，做参数高效的微调。
- `json_output.parser`：你项目里的一个自定义模块（脚本里没有直接使用到它，可能在其它地方用）。

------

**`2.LegalTextClassifierTrainer` 类 — 整个训练器的封装**

这个类把训练流程封装成了一个可复用的对象，主要职责：

- 初始化环境与资源（模型、分词器、标签映射）
- 把原始训练数据加载成列表
- 提供 `data_collator` 把样本拼成 batch
- 配置 LoRA 与训练参数
- 调用 `Trainer` 启动训练并在训练后保存模型

它的主要方法：`__init__` → `_load_resources` → `load_dataset` → `train`。

**3.初始化：`__init__` + `_load_resources`**

当你创建 `LegalTextClassifierTrainer(MODEL_PATH, ACCUSATION_MAP_PATH)` 时会做两件事：

1. `_setup_environment()`：当前函数留空（占位），表示可以放一些环境配置（比如禁用某些加速库或设定环境变量），目前没有实际操作。
2. `_load_resources()`：实际加载资源的地方：
   - 读取罪名（类别）映射文件 `accusation_map_path`，把罪名映射表载入为 `self.accusation_id`，并用它的长度来确定分类的 `num_labels`。
   - 使用 `AutoTokenizer.from_pretrained(self.model_path)` 从给定路径加载分词器（负责把文本转为 token id）。
   - 使用 `AutoModelForSequenceClassification.from_pretrained(...)` 加载一个用于分类的预训练模型，并把 `num_labels` 设为罪名数，这样模型最后一层输出的维度就是类别数。
   - 把模型配置里的 `pad_token_id` 设为 tokenizer 的 `pad_token_id`（这是为了后面 padding 时模型知道哪个 id 是填充符）。

简而言之：把“模型”和“分词器”与“分类类别数”都准备好，放到训练器对象里待用。

------

**4. `data_collator`：把若干样本合并成一个 batch**

`Trainer` 在训练时会给 `data_collator` 一个 batch（很多单条样本组成的列表），这个函数负责把这些单条样本变成模型能直接吃的张量（tensor）。

这个 `data_collator` 的做法是：

1. **先遍历一遍批次**，找出最长的 `input_ids`（最大序列长度），但会跳过长度超过 1000 的样本（认为太长就丢弃）。
2. **再遍历一次**，把每条样本的 `input_ids` 用 `pad_token_id` 右填充到步骤 1 得到的最大长度；同时读取样本里 `output` 字段作为标签 `label`。
3. 把填好的 `input_ids` 列表和 `labels` 列表分别变成 PyTorch 的 LongTensor，并用 `torch.stack` 堆成 `[batch_size, seq_len]` 的张量格式，返回一个包含 `"input_ids"` 和 `"labels"` 的字典。

通俗理解：把每条样本补齐长度，堆起来给模型训练用。注意这里没有返回 `attention_mask`，返回的是输入 id 和标签。

------

**5. LoRA 配置（`setup_lora_config`）**

LoRA（低秩适配器）是一种常用的参数高效微调方法，不用更新整个大模型的所有权重，只训练少量的 adapter 权重，节省显存与训练时间。

你的 LoRA 配置项含义：

- `r=16`：LoRA 的秩（越大灵活性越高，但参数越多）
- `lora_alpha=16`：缩放因子
- `target_modules=["q_proj","v_proj"]`：把 LoRA 附加在自注意力里的 q、v 投影上（常见选择）
- `lora_dropout=0.1`：LoRA 的 Dropout
- `bias="none"`：不训练偏置项
- `task_type=TaskType.SEQ_CLS`：表示这是序列分类的任务

返回一个 `lora_config`，后面传给 `get_peft_model` 来把 LoRA 注入模型。

------

**6. 训练参数（`setup_training_args`）**

这里用了 HuggingFace 的 `TrainingArguments`，主要配置包括：

- 输出目录 `output_dir`、学习率 `learning_rate`、每设备 batch size、梯度累积步数（用来虚拟增大 batch）、训练轮次 `num_train_epochs`、权重衰减 `weight_decay`。
- 日志目录 `logging_dir`、多少步记录一次日志 `logging_steps`。
- 保存策略 `save_strategy="steps"`，每 `save_steps=30` 保存一次检查点，最多保留 `save_total_limit=2` 个。
- `fp16=True`：启用混合精度（可以在支持的 GPU 上减小显存占用、提速）。
- `report_to="tensorboard"`：把训练过程上报给 TensorBoard，用于可视化。

这些参数决定训练时的调度、日志和保存行为。

------

**7. `load_dataset`：读取训练数据文件**

函数做的是把 `data_path` 指向的文件按行读入，每行是一个 JSON 字符串，`json.loads` 后包成 `{"input_ids": <json结构>}` 的字典，全部读成一个 Python 列表返回。

注意：这里的每行 JSON 的格式脚本没有明确写出，但 `data_collator` 期待的是 `feature["input_ids"]["input"]` 和 `feature["input_ids"]["output"]` 这两个字段，所以数据文件里的那条 JSON 应包含类似：

```json
{"input": [token_id,...], "output": label_int}
```

或对应的嵌套结构，最终被包成 `{"input_ids": {...}}`。

------

**8. `train`：训练主流程（一步步做的事）**

`train` 方法把训练的所有步骤串起来，主要流程如下：

1. 打印开始信息。
2. 调用 `setup_lora_config()` 得到 LoRA 配置，然后 `get_peft_model(self.model, lora_config)` 把 LoRA 注入到模型中。此时模型变成一个带 LoRA adapters 的可训练模型（大多数原始权重保持冻结，只训练少量 adapter 权重）。
3. 打印模型结构（`print(self.model)`）。
4. **解冻分类头的参数**：代码里通过 `for name, param in self.model.score.named_parameters()` 把分类器（score 层）参数的 `requires_grad` 设为 `True`，即允许训练这些参数。这一步目的通常是：在 LoRA 的基础上，同时微调最后的分类层（提高分类性能）。
5. 加载训练数据（`load_dataset`），并打印样本数量。
6. 获取训练参数（`setup_training_args`）。
7. 用 HuggingFace 的 `Trainer` 构建训练器，传入模型、训练参数、训练数据、以及自定义的 `data_collator`。
8. 调用 `trainer.train()` 启动训练过程（训练循环、日志、checkpoint）。
9. 训练结束后保存模型：
   - `self.model.save_pretrained("./qwen_lora_model")`：保存整个（或 LoRA）模型到指定目录。
   - `torch.save(self.model.score.state_dict(), "score_weights.pt")`：单独把分类头的权重保存为一个 `.pt` 文件。
10. 打印完成信息并返回 `trainer` 对象。

一句话：把 LoRA 注入模型，解冻分类头，加载数据，交给 Trainer 训练，训练完保存。

------

**9. `main()`：入口脚本**

`main()` 指定了三个路径：

- `MODEL_PATH`：预训练模型目录（例如本地 Qwen）。
- `ACCUSATION_MAP_PATH`：罪名映射文件路径（JSON）。
- `TRAIN_DATA_PATH`：训练数据文件路径。

用这些路径实例化 `LegalTextClassifierTrainer`，然后调用 `train()` 启动训练。

------

**10. 对你有用的“通俗提示”（运行时你应当知道的几件事）**

- **数据格式要对**：`data_collator` 期望输入字段 `input_ids.input` 与 `input_ids.output`（或等价的嵌套格式），数据文件里每行的 JSON 必须包含这些字段，才能正确组 batch。
- **LoRA 是为节省显存和训练成本**：不用更新整个大模型权重，只训练小部分 adapter，适合大模型在有限资源下微调。
- **`fp16=True` 能提升速度并节省显存**，但要求 GPU 与环境支持混合精度。
- **`Trainer` 会负责把数据搬到 GPU**，你自定义的 `data_collator` 返回 CPU tensor 即可（脚本里正是这样做的）。
- **保存部分**：脚本既保存了整个模型目录，也单独保存了分类头权重，便于后续复现或者只加载分类器权重。

```py
def build_model_and_tokenizer(base_model_path, lora_adapter_path=None, score_weights_path=None, device=None):

    """

    说明：
      - 使用 tokenizer(...) 生成 input_ids 与 attention_mask
      - 使用 torch.no_grad() 禁用梯度以节省显存
      - 返回最可能的标签（argmax）
    """
    # tokenize，返回 pytorch tensors（batch size = 1）

        text,
        truncation=True,
        max_length=max_length,
        return_tensors="pt"
    )

    # print("是否有掩码注意力：", attention_mask.shape)    # torch.Size([1, 197]) 见问题1

        attention_mask = attention_mask.to(device)

        # 某些 HF 分类模型返回 logits 形状为 (batch, num_labels)
        outputs = model(input_ids=input_ids, attention_mask=attention_mask)
        # print("模型输出：", outputs)
        logits = outputs.logits  # tensor shape (1, num_labels)
        # print("得分情况：", logits.shape)
        # print("得分：", logits)

        probs = torch.softmax(logits, dim=-1)[0]  # shape (num_labels,)

    # id_to_accusation: {int_id: label_name}
    return id_to_accusation.get(pred_idx, f"UNKNOWN_LABEL_{pred_idx}"), probs.cpu().tolist()
```

```py
def load_jsonl_files(paths):
    """把多份 jsonl 文件读成一个列表（每行 JSON）。"""
    lines = []
    for p in paths:
        with open(p, encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                obj = json.loads(line)
                lines.append(obj)
    return lines


def build_documents(raw_records, max_len=MAX_TEXT_LENGTH):
    """
    把原始记录转换为 langchain.Document 的列表。
    Document.page_content 存案情；metadata 包含判决结果与罪名（category）。
    """
    docs = []
    for rec in raw_records:
        text = rec.get("input", "")
        # text必须存在且长度 不能超过2000
        if not text or len(text) > max_len:
            continue

        # predict.accusation_json2str 将判决/结果对象转成可读字符串
        # 最后要变成字符串 才能进行embedding
        # output变成字符串
        result_str = predict.accusation_json2str(rec.get("output", {}))

        # 罪名可能是一个列表（多罪并罚），对每个罪名都创建一条文档，
        # 这样检索时可以按罪名过滤，且命中片段更精确
        #  {"是否死刑": False, "有期徒刑": 12, "是否无期": False, "罪名": ["故意伤害", "交通肇事"], "罚金": 0, "犯罪嫌疑人": ["段某"]}
        accusation_list = rec.get("output", {}).get("罪名", [])
        if not accusation_list:
            # 若没有罪名也保留文档，但 category 置为 None
            docs.append(Document(page_content=text, metadata={"result": result_str, "category": None}))
            continue
        # 若有罪名 每一个罪名 创建一条文档
        for accusation in accusation_list:
            docs.append(Document(page_content=text, metadata={"result": result_str, "category": accusation}))

    return docs


def build_faiss_index(documents, embeddings, batch_size=BATCH_SIZE, persist_path=None):
    """
    使用 LangChain 的 FAISS wrapper 建索引（批量计算 embedding）。
    返回构建好的 db（内存中的 FAISS 对象）。
    如果 persist_path 指定，会将对象 pickle 到磁盘（包含索引与映射）。
    """
    # LangChain 的 FAISS.from_documents 会内部计算 embeddings，但为了更好控制 batch 与进度，
    # 我们先把文档分成 batches，调用 embeddings.embed_documents 批量计算，再用 FAISS.from_documents 或 from_texts+metadatas 建库。
    texts = [d.page_content for d in documents]
    metadatas = [d.metadata for d in documents]
    # print(texts[:2])
    # print(metadatas[:2])
    all_texts = []  # 存储所有的input  长度和all_embeddings 一致
    # 计算 embeddings（批量）
    all_embeddings = []  # 所有的input所对应的embedding向量  512维度
    # batch_size = 128(默认)
    for i in tqdm(range(0, len(texts), batch_size), desc="Embedding batches"):
        batch_texts = texts[i:i + batch_size]
        # embeddings是一个对象
        batch_embs = embeddings.embed_documents(batch_texts)  # 期望返回 list[list[float]]
        # print(batch_embs)
        # print(len(batch_embs))  # 128
        # print(len(batch_embs[0]))  # 512
        all_texts.extend(batch_texts)
        all_embeddings.extend(batch_embs)

    # print(all_embeddings)
    text_embedding_pairs = zip(all_texts, all_embeddings)
    # 使用 FAISS 构建索引（从 embeddings + texts + metadatas）
    # LangChain 的 FAISS.from_embeddings 可直接接收 embeddings、texts、metadatas
    db = FAISS.from_embeddings(text_embeddings=text_embedding_pairs,
                               metadatas=metadatas,
                               embedding=embeddings
                               )
    # print(db)
    # 持久化（pickle 保存）——注意：pickle 会把整个索引与内存结构序列化
    if persist_path:
        with open(persist_path, "wb") as f:
            pickle.dump(db, f)

    return db

    return db

    """
    db: 向量数据库
    query: 需要检索的文本
    embeddings: 把query -->  embedding  -->  db  -->  5个最相似的文本向量

    返回 (docs_with_scores) 列表，每项包含 Document 与相似度分数（score 越大越相似）
    """

    return results
```
